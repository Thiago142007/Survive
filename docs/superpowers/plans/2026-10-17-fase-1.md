# Fase 1 — Fundação e Loop Básico — Plano de Implementação

> **Para agentes:** SUB-SKILL REQUERIDO: Use superpowers:subagent-driven-development (recomendado) ou superpowers:executing-plans para implementar tarefa por tarefa. Passos usam sintaxe de checkbox (`- [ ]`) para acompanhamento.

**Objetivo:** Estabelecer o projeto Godot 4 rodando a 60fps com movimentação em 1ª/2ª/3ª pessoa, grid de construção com fundação obrigatória, inventário com stack de 100 sem peso, e coleta básica de recursos.

**Arquitetura:** Projeto Godot 4 com renderer Compatibility. Jogador é um CharacterBody3D com câmera em 1ª/2ª/3ª pessoa. Sistema de grid para construções. Inventário como Array de dicionários. Recursos coletados via raycast de interação.

**Spec:** `docs/superpowers/specs/2026-10-17-survive-design.md`

## Restrições Globais

- Engine: Godot 4.x, renderer **Compatibility** (OpenGL 3.3)
- Linguagem: GDScript
- Resolução interna: **70%** da janela, escalonada (viewport stretch mode)
- Sem pós-processamento (bloom, SSAO, sombras dinâmicas)
- Sem limite de peso no inventário
- Stack máximo: **100 itens**
- Fundação é **obrigatória** antes de qualquer outra peça
- Grid de construção: células de **2m x 2m x 2m**
- Teclas: **R** = rotacionar eixo X (90°), **T** = rotacionar eixo Y (90°), **Shift+R/T** = 15°
- Perspectiva padrão: 1ª pessoa; tecla **V** alterna 1ª → 2ª → 3ª
- Hardware alvo: i5-2400, HD 5430, 2GB RAM livre

## Foco de Revisão

1. **Grid snapping** — peças devem se alinhar ao grid de 2m; sem alinhamento, o sistema de estresse (Fase 3) não funciona.
2. **Fundação obrigatória** — nenhuma peça pode ser colocada sem uma fundação conectada sob ela.
3. **Validação de colocação** — construir no ar deve ser bloqueado com mensagem no chat.
4. **Inventário sem peso** — stacks de 100; sem verificação de peso em nenhuma ação.
5. **60fps** — sem alocação por frame; sem sombras; sem pós-processamento.

- [ ] **Passo 1: Criar `scripts/player/player.gd`**

```gdscript
extends CharacterBody3D

@export var speed: float = 5.0
@export var sprint_speed: float = 8.0
@export var jump_velocity: float = 4.5
@export var mouse_sensitivity: float = 0.002

@onready var camera_pivot: Node3D = $CameraPivot
@onready var camera: Camera3D = $CameraPivot/Camera3D

var _cam_mode: int = 0  # 0=1ª, 1=2ª, 2=3ª
var _cam_distance: float = 0.0

func _ready() -> void:
    Input.mouse_mode = Input.MOUSE_MODE_CAPTURED
    Main.player = self

func _unhandled_input(event: InputEvent) -> void:
    if event is InputEventMouseMotion:
        rotate_y(-event.relative.x * mouse_sensitivity)
        camera_pivot.rotate_x(-event.relative.y * mouse_sensitivity)
        camera_pivot.rotation.x = clampf(camera_pivot.rotation.x, -PI/2, PI/2)
    if event.is_action_pressed("toggle_camera"):
        _toggle_camera()

func _toggle_camera() -> void:
    _cam_mode = (_cam_mode + 1) % 3
    match _cam_mode:
        0: _cam_distance = 0.0   # 1ª pessoa
        1: _cam_distance = 4.0   # 2ª pessoa
        2: _cam_distance = 6.0   # 3ª pessoa
    camera.position.z = _cam_distance

func _physics_process(delta: float) -> void:
    var input_dir := Input.get_vector("move_left", "move_right", "move_forward", "move_back")
    var direction := (transform.basis * Vector3(input_dir.x, 0, input_dir.y)).normalized()
    if direction:
        velocity.x = direction.x * speed
        velocity.z = direction.z * speed
    else:
        velocity.x = move_toward(velocity.x, 0, speed)
        velocity.z = move_toward(velocity.z, 0, speed)
    if is_on_floor() and Input.is_action_pressed("jump"):
        velocity.y = jump_velocity
    velocity.y -= 9.8 * delta
    move_and_slide()
```

- [ ] **Passo 2: Criar `scripts/inventory/inventory.gd`**

```gdscript
extends Node
class_name Inventory

var slots: Array[Dictionary] = []  # [{ "id": String, "qty": int, "data": ItemData }]
const MAX_STACK := 100

func add_item(id: String, qty: int, data: ItemData = null) -> bool:
    # Primeiro, tenta empilhar em slot existente
    for slot in slots:
        if slot["id"] == id and slot["qty"] < MAX_STACK:
            var space := MAX_STACK - slot["qty"]
            var to_add := mini(qty, space)
            slot["qty"] += to_add
            qty -= to_add
            if qty <= 0:
                return true
    # Depois, cria novos slots
    while qty > 0:
        var to_add := mini(qty, MAX_STACK)
        slots.append({ "id": id, "qty": to_add, "data": data })
        qty -= to_add
    return true

func remove_item(id: String, qty: int) -> bool:
    if get_count(id) < qty:
        return false
    var remaining := qty
    for i in range(slots.size() - 1, -1, -1):
        if slots[i]["id"] == id:
            var to_remove := mini(remaining, slots[i]["qty"])
            slots[i]["qty"] -= to_remove
            remaining -= to_remove
            if slots[i]["qty"] <= 0:
                slots.remove_at(i)
            if remaining <= 0:
                return true
    return true

func get_count(id: String) -> int:
    var total := 0
    for slot in slots:
        if slot["id"] == id:
            total += slot["qty"]
    return total

func get_all() -> Array[Dictionary]:
    return slots.duplicate()
```

**Interfaces:**

- [ ] **Passo 1: Criar `scripts/player/interaction.gd`**

```gdscript
extends Node3D
class_name Interaction

@export var raycast_length: float = 3.0
@onready var ray: RayCast3D = $RayCast3D

signal item_collected(id: String, qty: int)

func _ready() -> void:
    ray.force_raycast_update()

func _unhandled_input(event: InputEvent) -> void:
    if event.is_action_pressed("interact"):
        _try_interact()

func _try_interact() -> void:
    ray.force_raycast_update()
    if ray.is_colliding():
        var collider = ray.get_collider()
        if collider.has_method("harvest"):
            collider.harvest(self)

func register_resource(id: String, qty: int) -> void:
    var inv = Main.player.get_node("Inventory")
    inv.add_item(id, qty)
    item_collected.emit(id, qty)
```

- [ ] **Passo 2: Criar `scripts/building/building_system.gd`**

```gdscript
extends Node3D
class_name BuildingSystem

const GRID_SIZE := 2.0
var placed_pieces: Array[Node3D] = []
var ghost_piece: Node3D = null

signal piece_placed(piece: Node3D)
signal piece_removed(piece: Node3D)
signal invalid_placement(reason: String)

func snap_to_grid(pos: Vector3) -> Vector3:
    return Vector3(
        round(pos.x / GRID_SIZE) * GRID_SIZE,
        round(pos.y / GRID_SIZE) * GRID_SIZE,
        round(pos.z / GRID_SIZE) * GRID_SIZE
    )

func can_place(piece: BuildPiece, pos: Vector3, rot: Vector3) -> bool:
    var snapped := snap_to_grid(pos)
    # Verificar se já existe peça na mesma célula
    for p in placed_pieces:
        if p.position == snapped and p.rotation == rot:
            return false
    # Se não é fundação, verificar se há fundação conectada
    if not piece.is_foundation:
        var has_foundation_below := false
        var below_pos := snapped + Vector3(0, -GRID_SIZE, 0)
        for p in placed_pieces:
            if p.position == below_pos and (p as BuildPiece).is_foundation:
                has_foundation_below = true
                break
        if not has_foundation_below:
            invalid_placement.emit("Fundação necessária abaixo da peça")
            return false
    return true

func place_piece(piece: Node3D) -> void:
    piece.position = snap_to_grid(piece.position)
    placed_pieces.append(piece)
    piece_placed.emit(piece)

func remove_piece(piece: Node3D) -> void:
    placed_pieces.erase(piece)
    piece_removed.emit(piece)
    # Dropa 15% dos materiais
    var inv = Main.player.get_node("Inventory")
    for res in piece.build_cost:
        var drop_qty := int(ceil(piece.build_cost[res] * 0.15))
        if drop_qty > 0:
            inv.add_item(res, drop_qty)
    piece.queue_free()
```
